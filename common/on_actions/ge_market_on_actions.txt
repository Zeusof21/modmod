	on_monthly = {

            # ========================================================
            set_temp_variable = { var = ge_monthly_pool value = 30 }
            
            # Yıla göre aylık pazar kapasitesi artar
            if = { limit = { date > 1944.01.01 } set_temp_variable = { var = ge_monthly_pool value = 100 } }

            else_if = { limit = { date > 1940.01.01 } set_temp_variable = { var = ge_monthly_pool value = 60 } }

            # Eğer oyuncular stoğu tüketmişse, ay başında tekrar doldurulur!
            if = { limit = { check_variable = { var = global.ge_steel_pool value = ge_monthly_pool compare = less_than } } set_variable = { var = global.ge_steel_pool value = ge_monthly_pool } }
            if = { limit = { check_variable = { var = global.ge_oil_pool value = ge_monthly_pool compare = less_than } } set_variable = { var = global.ge_oil_pool value = ge_monthly_pool } }

            }
