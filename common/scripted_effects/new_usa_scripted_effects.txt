USA_victory_act_opinions = {
	custom_effect_tooltip = generic_skip_one_line_tt
	if = {
		limit = {
			NOT = {
				victory_act_taken = yes
			}
		}
		every_country = {
			limit = {
				NOT = {
					original_tag = USA
				}
				OR = {
					is_in_faction_with = USA
					has_war_with = USA
				}
			}
			add_opinion_modifier = {
				target = USA
				modifier = usa_victory_act_modifier
			}
		}
	}
	else = {
		every_country = {
			limit = {
				NOT = {
					original_tag = USA
				}
				OR = {
					is_in_faction_with = USA
					has_war_with = USA
				}
			}
			add_opinion_modifier = {
				target = USA
				modifier = usa_victory_act_more_modifier
			}
		}
	}
}

USA_improve_eco_dynamic_modifier = {
	hidden_effect = {
		if = {
			limit = {
				has_dynamic_modifier = USA_economy_dynamic_modifier_great_depression
			}
			remove_dynamic_modifier = {
				modifier = USA_economy_dynamic_modifier_great_depression
			}
			add_dynamic_modifier = {
				modifier = USA_economy_dynamic_modifier_good
			}
		}
		else_if = {
			limit = {
				has_dynamic_modifier = USA_economy_dynamic_modifier_bad
			}
			remove_dynamic_modifier = {
				modifier = USA_economy_dynamic_modifier_bad
			}
			add_dynamic_modifier = {
				modifier = USA_economy_dynamic_modifier_good
			}
		else_if = {
			limit = {
				has_dynamic_modifier = USA_economy_dynamic_modifier_good
			}
			remove_dynamic_modifier = {
				modifier = USA_economy_dynamic_modifier_good
			}
			add_dynamic_modifier = {
				modifier = USA_economy_dynamic_modifier_great
			}
		}
		}
	}
}
