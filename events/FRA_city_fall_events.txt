
add_namespace = city_fall_news

# Bordeaux
news_event = {
	id = city_fall_news.8
	title = city_fall_news.8.t
	desc = city_fall_news.8.d
	picture = GFX_news_event_city_fall_bordeaux
	
	major = yes
	is_triggered_only = yes
	
	trigger = {
		FROM = { has_capitulated = no }
	}
	
	immediate = {
		19 = {
			set_state_flag = state_has_fallen
		}
		FROM.FROM = { save_event_target_as = old_controller }
		ROOT = { save_event_target_as = new_controller }		
	}

	option = {
		name = city_fall_news.8.a
		trigger = {
			tag = event_target:old_controller
		}
	}
	option = {
		name = city_fall_news.8.b
		trigger = {
			tag = event_target:new_controller
		}
	}
	option = {
		name = city_fall_news.8.c
		trigger = {
			NOT = {
				tag = event_target:old_controller
				tag = event_target:new_controller
			}
		}
	}
}
